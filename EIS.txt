    "def parse_dates_robustly(df, col_name=\"Invoice_Date\"):\n",
    "    \"\"\"\n",
    "    Safely parses datetime strings. Features:\n",
    "    - Preserves raw data for auditing\n",
    "    - Handles mixed formats (pandas 2.0+ compatible)\n",
    "    - Assumes day-first formats commonly seen in Excel exports\n",
    "    - Validates and logs exact failure counts\n",
    "    \"\"\"\n",
    "    if col_name not in df.columns:\n",
    "        return df\n",
    "        \n",
    "    df = df.copy()\n",
    "    raw_col = f\"{col_name}_raw\"\n",
    "    df[raw_col] = df[col_name]  # Preserve original for auditing\n",
    "    \n",
    "    # Clean string artifacts (strip spaces) if the data type is string\n",
    "    if df[col_name].dtype == 'object':\n",
    "        df[col_name] = df[col_name].apply(lambda x: x.strip() if isinstance(x, str) else x)\n",
    "\n",
    "    # Defensive Parsing\n",
    "    try:\n",
    "        # 'mixed' format flag works in Pandas 2.0+ to handle varying structures\n",
    "        df[col_name] = pd.to_datetime(df[col_name], errors='coerce', format='mixed', dayfirst=True)\n",
    "    except ValueError:\n",
    "        # Fallback for older pandas versions\n",
    "        df[col_name] = pd.to_datetime(df[col_name], errors='coerce', dayfirst=True)\n",
    "\n",
    "    # Validation Layer\n",
    "    # Identify rows that *became* NaT during conversion but weren't null originally\n",
    "    failed_mask = df[col_name].isna() & df[raw_col].notna()\n",
    "    failed_count = failed_mask.sum()\n",
    "\n",
    "    if failed_count > 0:\n",
    "        logging.warning(f\"[{col_name}] {failed_count} non-null items failed conversion and are now NaT.\")\n",
    "        print(\"Sample of failed conversions:\")\n",
    "        display(df.loc[failed_mask, [raw_col, col_name]].head())\n",
    "    else:\n",
    "        logging.info(f\"[{col_name}] All valid inputs parsed successfully.\")\n",
    "\n",
    "    return df\n",
    "\n",
    "# Apply the transformation pipeline\n",
    "for name in dfs.keys():\n",
    "    dfs[name] = parse_dates_robustly(dfs[name])"
